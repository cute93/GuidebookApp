observeViewModel() 교체 패턴

Kotlin
Kotlin 
private val participantSlots = listOf(
    "teacher"  to Triple(binding.tvTeacherName,  binding.ivTeacherNote,  binding.btnTeacherWrite),
    "student1" to Triple(binding.tvStudent1Name, binding.ivStudent1Note, binding.btnStudent1Write),
    "student2" to Triple(binding.tvStudent2Name, binding.ivStudent2Note, binding.btnStudent2Write),
    "student3" to Triple(binding.tvStudent3Name, binding.ivStudent3Note, binding.btnStudent3Write)
)

viewModel.notes.observe(this) { notes ->
    participantSlots.forEach { (role, views) ->
        val (tvName, ivNote, btnWrite) = views
        val note = notes.find { it.role == role }
        tvName.text = note?.userName ?: role
        if (note?.noteImageUrl != null) {
            Glide.with(this).load(note.noteImageUrl).into(ivNote)
        }
        val canEdit = note?.userId == currentUser.uid ||
            (currentUser.role == "teacher" && role == "teacher")
        btnWrite.isEnabled = canEdit
        btnWrite.setOnClickListener {
            if (canEdit) navigateToPage2(note)
        }
    }
}

Kotlin 
// 저장 경로: filesDir/drafts/{problemId}_{userId}.png
private fun saveDraftLocally() {
    val dir = File(filesDir, "drafts").also { it.mkdirs() }
    val file = File(dir, "${problemId}_${currentUser.uid}.png")
    file.outputStream().use { out ->
        binding.drawingView.getBitmap()
            .compress(Bitmap.CompressFormat.PNG, 100, out)
    }
    Toast.makeText(this, "임시 저장되었습니다.", Toast.LENGTH_SHORT).show()
}

private fun loadDraftIfExists() {
    val file = File(filesDir, "drafts/${problemId}_${currentUser.uid}.png")
    if (file.exists()) {
        val bmp = BitmapFactory.decodeFile(file.absolutePath)
        binding.drawingView.loadBitmap(bmp)
    }
}

// onCreate에서 호출
loadDraftIfExists()
binding.btnSaveDraft.setOnClickListener { saveDraftLocally() }

DrawingView.kt — loadBitmap() 추가

views/DrawingView.kt
Kotlin 
// 기존 getBitmap()과 대칭되는 메서드
fun loadBitmap(bitmap: Bitmap) {
    canvas.drawBitmap(bitmap, 0f, 0f, null)
    invalidate()
}
ℹ️

DrawingView는 오프스크린 Bitmap에 렌더링하므로, 로드된 비트맵을 canvas에 직접 그린 뒤 invalidate()로 화면 갱신.
